## Solution

class Solution {
public:
    vector<int> exclusiveTime(int n, vector<string>& logs) {

        stack<string> st;
        map<int, int> mp;

        for (const string& it : logs) {
            stringstream cur_log(it);
            string id_str, action, timestamp_str;

            getline(cur_log, id_str, ':');
            getline(cur_log, action, ':');
            getline(cur_log, timestamp_str, ':');

            int cur_id = stoi(id_str);
            int cur_timestamp = stoi(timestamp_str);

            if (action == "start") {
                st.push(it);
            } else { //if the log is of type "end"
                stringstream top_log(st.top());
                string top_id_str, top_action, top_timestamp_str;

                getline(top_log, top_id_str, ':');
                getline(top_log, top_action, ':');
                getline(top_log, top_timestamp_str, ':');

                int top_id = stoi(top_id_str);
                int top_timestamp = stoi(top_timestamp_str);

                int cur_duration = (cur_timestamp - top_timestamp + 1);
                mp[cur_id] += cur_duration;

                st.pop(); // removing the top log

                if (!st.empty()) { // if cur function call happens in the middle
                                   // of the other one then, the time of already
                                   // existing will be taken by the cur function
                                   // call, hence will reduce the time of the
                                   // other by cur_duration
                    stringstream ontop_log(st.top());
                    string ontop_id_str, ontop_action, ontop_timestamp_str;

                    getline(ontop_log, ontop_id_str, ':');
                    int ontop_id = stoi(ontop_id_str);

                    mp[ontop_id] -= cur_duration;
                }
            }
        }
        vector<int> times;
        for (const auto& [id, duration] : mp) {
            times.push_back(duration);
        }
        return times;
    }
};
