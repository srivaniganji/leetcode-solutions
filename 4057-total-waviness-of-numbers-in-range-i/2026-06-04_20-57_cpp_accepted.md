## Solution

```cpp
class Solution {
public:
    int getWaviness(string s){
        if(s.size()<=2)  return 0;
        int waviness = 0;
        for(int i=1;i<s.size()-1;i++){
            int l = s[i-1]-'0', c = s[i]-'0', r = s[i+1]-'0';
            if((c>l && c>r) || (c<l && c<r))  waviness++;
        }
        return waviness;
    }
    int totalWaviness(int num1, int num2) {
        int ans = 0;
        for(int i=num1;i<=num2;i++){
            ans += getWaviness(to_string(i));
        }
        return ans;
    }
    
};
```
