## Solution

```java
class Solution {
    public int[] pivotArray(int[] nums, int pivot) {
        int n = nums.length;
        int[] less = new int[n];
        int[] greater = new int[n];
        int[] equal = new int[n];
        int l = 0, g = 0, e = 0;
        for(int num: nums){
            if(num<pivot)   less[l++] = num;
            else if(num==pivot) equal[e++] = num;
            else    greater[g++] = num;
        }
        int[] result = new int[n];
        int k = 0;
        for(int i=0;i<l;i++)    result[k++] = less[i];
        for(int i=0;i<e;i++)    result[k++] = equal[i];
        for(int i=0;i<g;i++)    result[k++] = greater[i];
        return result;
    }
}
```
